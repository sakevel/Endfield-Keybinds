cmake_minimum_required(VERSION 3.25)
project(EndfieldKeybinds VERSION 0.2.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(Keybinds SHARED src/plugin.cpp)
target_include_directories(Keybinds PRIVATE sdk/include src)
set_target_properties(Keybinds PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/keybinds")
add_custom_target(package ALL DEPENDS Keybinds)
foreach(file mod.ini keybinds.lua platform.lua settings.lua zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${file}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/keybinds/${file}")
endforeach()
enable_testing()
add_library(ServiceFixture SHARED tests/service_fixture.cpp)
target_include_directories(ServiceFixture PRIVATE sdk/include)
set_target_properties(ServiceFixture PROPERTIES OUTPUT_NAME ZMLRuntime RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>")
add_executable(NativeTests tests/native_tests.cpp)
target_include_directories(NativeTests PRIVATE sdk/include)
add_test(NAME native_api COMMAND NativeTests "$<TARGET_FILE:ServiceFixture>" "$<TARGET_FILE:Keybinds>")
add_executable(PluginTests tests/plugin_tests.cpp)
target_include_directories(PluginTests PRIVATE sdk/include src)
add_test(NAME plugin COMMAND PluginTests "$<TARGET_FILE:Keybinds>")
set(ZML_LUPA_PATH "" CACHE PATH "Existing Lupa directory, no download")
if(ZML_LUPA_PATH)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME lua_behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/behavior.py" --lupa-dir "${ZML_LUPA_PATH}")
  set_tests_properties(lua_behavior PROPERTIES TIMEOUT 30)
endif()
set(ZML_CLIENT_LUA "" CACHE PATH "Optional freshly extracted client Lua (ignored local evidence)")
if(ZML_CLIENT_LUA)
  add_test(NAME client_contract COMMAND PluginTests "$<TARGET_FILE:Keybinds>" "${ZML_CLIENT_LUA}" "${CMAKE_BINARY_DIR}/client-patched")
  if(ZML_LUPA_PATH)
    add_test(NAME client_behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/client.py"
      --lupa-dir "${ZML_LUPA_PATH}" --client "${ZML_CLIENT_LUA}" --patched "${CMAKE_BINARY_DIR}/client-patched")
    set_tests_properties(client_behavior PROPERTIES DEPENDS client_contract TIMEOUT 30)
  endif()
endif()
